# ======================================================================
# PYTHON DEPENDENCIES
# ======================================================================
# Install everything with:   pip install -r requirements.txt
#
# Every version is PINNED with `==` rather than left open. This means everyone
# who installs this project gets byte-for-byte the same library versions, so
# code that works on your laptop cannot mysteriously break on the server
# because a dependency shipped a new release overnight.
# ======================================================================

# ---- WEB FRAMEWORK ---------------------------------------------------
# fastapi  : the framework itself -- routing, validation, auto-generated docs.
# uvicorn  : the ASGI server that actually listens on a port and runs FastAPI.
#            FastAPI defines WHAT happens; uvicorn is what runs it.
fastapi==0.143.3
uvicorn[standard]==0.54.0

# `python-multipart` is required for file uploads. Without it, FastAPI cannot
# parse the multipart/form-data encoding a browser uses to send files, and the
# upload endpoint fails at import time with a clear error.
python-multipart==0.0.32

# ---- CONFIGURATION AND VALIDATION -----------------------------------
# pydantic          : validates and converts data (all our schemas).
# pydantic-settings : reads settings from environment variables and .env.
# python-dotenv     : the underlying .env file parser.
pydantic==2.14.1
# EmailStr (app/schemas/user.py) needs this. FastAPI 0.111 happened to install
# it; from 0.112 it does not, so it has to be listed directly.
email-validator==2.3.0
pydantic-settings==2.16.0
python-dotenv==1.2.4

# ---- DATABASE --------------------------------------------------------
# sqlalchemy      : the ORM -- write Python objects instead of raw SQL.
# asyncpg         : the ASYNC Postgres driver, used by the FastAPI app.
# psycopg2-binary : the SYNC Postgres driver, needed by Alembic migrations.
# alembic         : applies versioned schema changes to the database.
# pgvector        : adds the `Vector` column type and distance operators.
sqlalchemy[asyncio]==2.1.4
asyncpg==0.32.0
psycopg2-binary==2.9.13
alembic==1.20.0
pgvector==0.5.1

# ---- AUTHENTICATION AND SECURITY -------------------------------------
# python-jose : creates and verifies the signed JWT login tokens.
# passlib     : the password hashing framework.
# bcrypt      : the actual hashing algorithm. It is deliberately SLOW, which is
#               what makes brute-forcing a stolen password database impractical.
python-jose[cryptography]==3.5.0
passlib[bcrypt]==1.7.4
bcrypt==4.3.0

# ---- BACKGROUND JOBS -------------------------------------------------
# celery : runs slow work (document processing) outside the web request, so the
#          user gets an instant response instead of waiting 30 seconds.
# redis  : the message queue Celery uses to pass jobs to its workers.
celery==5.6.3
redis==8.1.0

# ---- DOCUMENT PARSING (one library per format family) ----------------
# pymupdf        : PDF -- text, page count, metadata, and page rasterisation.
# python-docx    : Word .docx -- paragraphs AND tables.
# python-pptx    : PowerPoint .pptx -- text from every shape on every slide.
# openpyxl       : Excel .xlsx/.xls -- cell values, sheet by sheet.
# beautifulsoup4 : HTML -- strips tags and returns the readable text.
# langdetect     : identifies which human language a document is written in.
# (.eml, .csv, .zip and .txt are handled by Python's own standard library.)
pymupdf==1.28.2
python-docx==1.2.0
python-pptx==1.0.2
openpyxl==3.1.5
beautifulsoup4==4.15.0
langdetect==1.0.9

# ---- OCR (reading text out of scanned pages and images) --------------
# pytesseract : a thin Python wrapper around the Tesseract OCR program.
#               NOTE: the Tesseract PROGRAM must also be installed separately --
#               it is not a Python package. See docs/DEVELOPMENT.md for instructions.
# pillow      : image loading and manipulation, which pytesseract needs.
pytesseract==0.3.13
pillow==12.3.0

# ---- AI: EMBEDDINGS AND TEXT GENERATION ------------------------------
# tiktoken : counts tokens exactly the way the models do, so our chunks are
#            genuinely 500 tokens rather than 500 characters.
tiktoken==0.14.0

# sentence-transformers : runs the embedding model LOCALLY on this server.
#            This is our default provider: free, no API key, works offline, and
#            document text never leaves the machine.
#            It pulls in PyTorch, which is why installation is a large download.
sentence-transformers==6.1.0

# groq : client for Groq's API, which serves open models such as
#            openai/gpt-oss-120b (our default, see GROQ_MODEL) and writes the
#            final answers. Free tier, no credit card.
#            >>> Get your key at https://console.groq.com/keys and put it in
#            >>> backend/.env as   GROQ_API_KEY=gsk_...
groq==1.7.0

# pydantic-ai : the agent framework behind chat "agent" mode
#            (app/services/agent.py). It runs the tool-calling loop for us:
#            turns our typed Python functions into tools the model can call,
#            validates the arguments the model sends back, and stops the loop
#            at a usage limit. Same team as pydantic, which we already use.
#
#            `-slim` installs ONLY the core plus the providers you name in the
#            brackets. The full `pydantic-ai` package pulls in the SDK of every
#            supported AI provider (OpenAI, Anthropic, Google, Mistral, ...),
#            none of which we use. `[groq]` adds Groq support, which reuses the
#            `groq` package pinned above (it needs groq>=0.25).
#
#            Optional at runtime: if it is missing, the app still starts and
#            agent mode falls back to its keyless demo answer.
pydantic-ai-slim[groq]==1.107.9

# openai : OPTIONAL. Only needed if you set EMBEDDING_PROVIDER=openai to use
#            paid OpenAI embeddings instead of the free local model.
openai==3.31.0

# ---- TESTING ---------------------------------------------------------
# Needed only to RUN the test suite, never by the application itself.
# They are listed here rather than in a separate requirements-dev.txt so that
# `pip install -r requirements.txt` leaves you able to run `pytest` -- a suite
# nobody can run is a suite nobody runs.
#
#     python -m pytest                    everything
#     SKIP_DB_TESTS=1 python -m pytest    Tier 1 only, no database needed
pytest==9.1.1

# pytest-asyncio : lets `async def test_...` run at all. The Tier 2 tests await
#            real database calls, and without this they would be collected,
#            reported as passing, and never actually executed.
pytest-asyncio==1.4.0

# httpx : used by tests/test_api.py to make real HTTP requests against the app
#            through an in-process ASGI transport -- no uvicorn and no network
#            port, so the tests are faster and cannot flake on a busy socket.
#            (It also arrives as a dependency of the groq and openai clients;
#            it is pinned here because the tests import it directly.)
httpx==0.28.1
